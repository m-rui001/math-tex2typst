% \rd{name}\macro: the anchor of a named node, in pt, read straight out of pgf. This is the point
% TeX draws at; the converter's canvas has to agree with it after the cm/pt conversion.
\def\rd#1#2{\pgfextractx{\mx}{\pgfpointanchor{#1}{center}}%
            \pgfextracty{\my}{\pgfpointanchor{#1}{center}}%
            \edef#2{\the\mx,\the\my}}
\begin{document}
% Absolute positions the strip's nine cells land on. The strand node (t) is seeded off-canvas so
