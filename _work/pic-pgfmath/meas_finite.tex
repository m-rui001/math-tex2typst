   \pgfmathsetmacro{\cy}{\j+.35*\i}

   \pgfmathsetmacro{\ly}{\cy-.5}
   \pgfmathsetmacro{\hx}{\cx+.5}
   \pgfmathsetmacro{\hy}{\cy+.5}
   \immediate\write\meas{CELL i=\i\space j=\j\space cx=\cx\space cy=\cy\space lo=\lx,\ly\space hi=\hx,\hy}
 }}
