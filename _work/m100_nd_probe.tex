% One dump per (reference node, new node) pair. Each anchor is read as a zero-length lineto so
% \pgfgetlastxy has a last position on the path; it answers in the picture's user coordinates (轴 65).
\def\dump#1#2#3#4{%
  \path (#1.east) -- +(0pt,0pt);  \pgfgetlastxy{\AdiA}{\AdiB}%
  \path (#1.north) -- +(0pt,0pt); \pgfgetlastxy{\AniA}{\AniB}%
  \path (#1.center) -- +(0pt,0pt);\pgfgetlastxy{\AciA}{\AciB}%
  \path (#2.west) -- +(0pt,0pt);  \pgfgetlastxy{\BwiA}{\BwiB}%
  \path (#2.south) -- +(0pt,0pt); \pgfgetlastxy{\BsiA}{\BsiB}%
  \path (#2.center) -- +(0pt,0pt);\pgfgetlastxy{\BciA}{\BciB}%
  % one record per anchor: \typeout breaks a log line at TeX's \line, which LaTeX shadows with a
  % macro, so a record cannot be widened -- it has to be written short
  \typeout{DUMP|#3|#4|A.east|\AdiA|\AdiB}%
  \typeout{DUMP|#3|#4|A.north|\AniA|\AniB}%
  \typeout{DUMP|#3|#4|A.center|\AciA|\AciB}%
  \typeout{DUMP|#3|#4|B.west|\BwiA|\BwiB}%
  \typeout{DUMP|#3|#4|B.south|\BsiA|\BsiB}%
  \typeout{DUMP|#3|#4|B.center|\BciA|\BciB}%
}
\makeatother
